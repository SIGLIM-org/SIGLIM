-- Suma independiente de la API: PENDIENTE + VENCIDO por club y equipo (ROLLUP da subtotal por club y total de la liga).
SELECT coalesce(cl.nombre, '== TOTAL LIGA ==') AS club,
       CASE WHEN cl.nombre IS NULL THEN '' WHEN GROUPING(e.id) = 1 THEN '== total del club ==' ELSE coalesce(e.nombre, '(cargo al club, sin equipo)') END AS equipo,
       sum(c.monto) AS total,
       sum(c.monto) FILTER (WHERE c.estado = 'PENDIENTE') AS pendiente,
       coalesce(sum(c.monto) FILTER (WHERE c.estado = 'VENCIDO'), 0.00) AS vencido,
       count(*) AS cobros
  FROM siglim.cobros c
  JOIN siglim.clubes cl ON cl.id = c.club_id
  LEFT JOIN siglim.equipos e ON e.id = c.equipo_id
 WHERE c.estado IN ('PENDIENTE', 'VENCIDO')
 GROUP BY ROLLUP (cl.nombre, (e.id, e.nombre))
 ORDER BY cl.nombre NULLS LAST, GROUPING(e.id), e.nombre NULLS LAST;

-- Cobros excluidos del adeudo (deben seguir existiendo y listarse).
SELECT cl.nombre AS club, e.nombre AS equipo, c.concepto, c.monto, c.estado
  FROM siglim.cobros c JOIN siglim.clubes cl ON cl.id = c.club_id LEFT JOIN siglim.equipos e ON e.id = c.equipo_id
 WHERE c.estado NOT IN ('PENDIENTE', 'VENCIDO');
